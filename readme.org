** use-package
use-packageを有効化する
melpaはstableを優先する
最新パッケージの場合、NT Emacsではうまく動かない場合がある
#+begin_src emacs-lisp
  (eval-and-compile
    (require 'package)
    ;; package-vcで導入したパッケージをproject.elへ登録しない
    (setq package-vc-register-as-project nil)

    (package-initialize)
    (add-to-list 'package-archives '("melpa-stable" . "https://stable.melpa.org/packages/") t)
    (add-to-list 'package-archives '("melpa" . "https://melpa.org/packages/") t)
    (add-to-list 'package-archives '("gnu" . "https://elpa.gnu.org/packages/") t)

    ;; set package-archives priority
    (setq package-archive-priorities
          '(("gnu" . 20)
            ("melpa" . 10)
            ("melpa-stable" . 5)))

    ;; 組み込みパッケージの上書きアップグレードを無効化
    (setq package-install-upgrade-built-in nil)
    (setq package-native-compile t)

    ;; use-package
    (unless (package-installed-p 'use-package)
      (package-refresh-contents)
      (package-install 'use-package))

    ;; ensureは基本nil. 必要なパッケージは個別でtにする
    (setq use-package-always-ensure nil)
    (setq use-package-expand-minimally t)
    (setq use-package-compute-statistics t) ;; For "M-x use-package-report"

    ;; nerd-iconsはmelpaから取得する
    ;; nerd-icons-*関連パッケージがmelpaにしかなく、melpa-stableのnerd-iconsでは動かない場合がある
    (add-to-list 'package-pinned-packages '(nerd-icons . "melpa"))

    (require 'use-package))
#+end_src

** Optimizing performance
**** プロセスからの出力の受信
[[https://ayatakesi.github.io/lispref/25.2/html/Output-from-Processes.html]]
いくつかのシステムではEmacsがサブプロセスの出力を読み取る際に出力データを非常に小さいブロックで読み取るために、
結果として潜在的に非常に貧弱なパフォーマンスとなることがある。
この挙動は変数process-adaptive-read-bufferingを非nil値(デフォルト)にセットして拡張することにより改善し得る。
これにより、そのようなプロセスからの読み取りを自動的に遅延して、Emacsが読み取りを試みる前に出力がより多く生成されるようになる。
#+begin_src emacs-lisp
  (setq process-adaptive-read-buffering t)
#+end_src

    ;; MSYS2に同梱されているgpgを用いると、WindowsパスがC:\Windows~として扱われてしまう
    ;; これによって署名を認識できず、packageの更新が一切行えなくなる

    ;; 現在は.bash_profileに「export PATH="/c/Program Files/GnuPG/bin":$PATH」を書いて起動している
    ;; ref : https://www.youtube.com/watch?v=9ha4AkqUPp4
    ;;       https://www.reddit.com/r/emacs/comments/ymzw78/windows_elpa_gnupg_and_keys_problems/
    (make-directory package-gnupghome-dir t)

    (defun my/apply-gui-frame-settings (&optional frame)
      (let ((frame (or frame (selected-frame))))
        (when (display-graphic-p frame)
          (with-selected-frame frame
            ;; GUI整理
            (tool-bar-mode 1)
            (scroll-bar-mode 1)
            (menu-bar-mode 1)

            ;; IDEっぽいタイトル
            (setq frame-title-format
                  '(:eval
                    (format "%s — Emacs"
                            (abbreviate-file-name
                             (or (buffer-file-name)
                                 (buffer-name))))))

            ;; ウィンドウ区切り線
            (setq window-divider-default-right-width 1
                  window-divider-default-bottom-width 1)
            (window-divider-mode 1)))))

    ;; 起動後の既存フレーム
    (add-hook 'emacs-startup-hook #'my/apply-gui-frame-settings)

    ;; 新規フレーム（daemon対応）
    (add-hook 'after-make-frame-functions #'my/apply-gui-frame-settings)
    
    :custom
    ;; --- Bell / Beep 無効化---
    (ring-bell-function #'ignore)
    ;; 画面フラッシュ
    (visible-bell nil)
    ;; emacsデフォルトのスタートアップ画面表示を止める
    (inhibit-startup-screen t)
    ;; 非選択ウィンドウではカーソルを表示しない
    (cursor-in-non-selected-windows nil)
    ;; 選択した範囲の文字を削除し、入力文字に置き換える
    (delete-selection-mode t)
    ;; コンテキストメニュー表示を許可する
    (context-menu-mode t)
    
    ;; --- スクロール改善 ---
    (scroll-conservatively 101)
    (scroll-step 1)
    ;; pixel-scroll が滑らかになるらしいのだが、visual-line-modeと相性が悪い
    ;; またマウス専用の仕組みみたいで、キーボードでのスクロールには向かない
    (pixel-scroll-precision-mode -1)
    (pixel-scroll-precision-interpolate-page -1)
    ;; --- ネイティブコンパイル警告抑制---
    (native-comp-async-report-warnings-errors 'silent)
    ;;  ---  行番号  --- 
    (display-line-numbers-type 'absolute)  ; 絶対行番号で表示する
    (display-line-numbers-width-start t)   ; 行番号の幅を最初から確保する
    :config
    ;; --- y-or-n-p ---
    (defalias 'yes-or-no-p 'y-or-n-p)
    ;; y-or-n-p を Return で確定できるようにする
    (advice-add 'y-or-n-p :around
                (lambda (orig-func &rest args)
                  (let ((query-replace-map (copy-keymap query-replace-map)))
                    (keymap-set query-replace-map "<return>" 'act)
                    (apply orig-func args))))
    ;; --- Windows 最適化---
    (when (eq system-type 'windows-nt)
      (setopt w32-get-true-file-attributes nil   ; ファイルI/O負荷軽減
              w32-use-native-image-API t         ; 画像APIをWindowsネイティブに

    ;; --- show-paren を使ってかっこを強調表示させる special-mode/text-mode/hexl-modeでは抑制する---

            show-paren-when-point-inside-paren t
            show-paren-predicate
            (lambda (&rest _args)
              (not (or (derived-mode-p 'special-mode)
                       (derived-mode-p 'text-mode)
                       (derived-mode-p 'hexl-mode)))))
    (show-paren-mode 1)
    ;; 大文字小文字変換を許可

      (let ((target

        (if (and (window-live-p target)

    (defun my/dired-open ()
      "Open the current Dired item.

  Directories are opened in the same Dired buffer.
  Files are opened normally, except from Side Dired where they are
  opened in the associated editing window."
      (interactive)
      (let* ((window (selected-window))
             (file (dired-get-file-for-visit))
             (side-p (window-parameter window 'my-dired-side)))
        (cond
         ;; ディレクトリは常に同じDiredバッファで移動
         ((file-directory-p file)
          (dired-find-alternate-file))

         ;; Side Diredでは右側の編集ウィンドウで開く
         (side-p
          (let ((target (my/dired-side-target-window window)))
            (unless target
              (user-error "No editing window available"))
            (set-window-buffer target (find-file-noselect file))
            (select-window target)))

         ;; 通常Diredでは普通に開く
         (t
          (dired-find-file)))))

    "Prevent eglot-java from destructively modifying JDTLS JVM arguments."
    (let ((eglot-java-eclipse-jdt-args
           (copy-sequence eglot-java-eclipse-jdt-args)))
      (apply orig-fun args)))

  (use-package eglot-java
    :ensure t
    :demand t
    :config
    ;; eglot-java--eclipse-jdt-contact 内の破壊的リスト変更を回避
    (unless (advice-member-p
             #'my/eglot-java-preserve-jdt-args
             'eglot-java--eclipse-jdt-contact)
      (advice-add
       'eglot-java--eclipse-jdt-contact
       :around
       #'my/eglot-java-preserve-jdt-args))

    ;; java-ts-mode 用インデント
    (setq-default java-ts-indent-offset 2)

    ;; 通常 java-mode 用インデント
    (setq-default c-basic-offset 2)

    ;; Java var の推論型を Inlay Hint で表示

  ;; Windowsに限りcase-insensitiveなfallbackを追加する。

      (orig-fun server abspath)
    "Find Eglot managed buffer case-insensitively on Windows."
    (or
     (funcall orig-fun server abspath)
     (when IS-WINDOWS
       (cl-loop
        for buffer in (eglot--managed-buffers server)
        when
        (with-current-buffer buffer
          (let ((cached
                 (car-safe eglot--TextDocumentIdentifier-cache)))
            (or
             (and buffer-file-name
                  (string-equal-ignore-case
                   abspath buffer-file-name))
             (and cached
                  (string-equal-ignore-case
                   abspath cached)))))
        return buffer))))

*** internal helper functions
各出力関数が共通で使う内部関数。
pandoc実行、出力ファイル名の生成、PDF専用引数の組み立て、出力後にファイルを開く処理をまとめている。
#+begin_src emacs-lisp
  ;; pandocを実行する共通関数
  ;; 失敗した場合はpandocの標準出力・標準エラーをそのまま表示して原因を追えるようにする
  (defun my/pandoc--run (args)
    "Run pandoc with ARGS.
  Signal an error with pandoc output if the command fails."
    (with-temp-buffer
      (let ((exit-code (apply #'process-file my/pandoc-command nil t nil args))
            (output (buffer-string)))
        (if (zerop exit-code)
            t
          (error "pandoc failed:\n%s" output)))))

  ;; 出力ファイル名を生成する共通関数
  ;; 例えば sample.org -> sample.pdf のように拡張子だけを置き換える
  (defun my/pandoc--target-file (src ext)
    "Return output filename from SRC with extension EXT."
    (concat (file-name-sans-extension src) "." ext))

  ;; PDF出力時の共通引数
  ;; PDF関係の設定を一箇所へまとめることで、後からエンジンや余白を変えやすくする
  (defun my/pandoc--pdf-args ()
    "Return common pandoc arguments for PDF export."
    (append
     (list "--pdf-engine" my/pandoc-pdf-engine
           "-V" (format "mainfont=%s" my/pandoc-mainfont)
           "-V" "papersize=a4"
           "-V" (format "geometry:%s" my/pandoc-pdf-geometry))
     (when my/pandoc-pdf-template
       (list "--template" my/pandoc-pdf-template))))

  ;; 出力後に外部アプリでファイルを開く補助関数
  ;; Windows, macOS, Linux のそれぞれで既定アプリを呼ぶ
  (defun my/pandoc--open-file (file)
    "Open FILE with the default external application."
    (cond
     ((eq system-type 'windows-nt)
      (w32-shell-execute "open" (expand-file-name file)))
     ((eq system-type 'darwin)
      (process-file "open" nil 0 nil file))
     (t
      (process-file "xdg-open" nil 0 nil file))))

  ;; 現在のOrgバッファからタイトル文字列を取得する
  ;; #+TITLE: が無ければ nil を返す
  (defun my/pandoc--org-title ()
    "Return current Org buffer title string, or nil if none."
    (when (derived-mode-p 'org-mode)
      (let* ((title (org-collect-keywords '("TITLE")))
             (raw   (cadar title)))
        (when (and raw (not (string-empty-p raw)))
          raw))))

  ;; Markdownファイルの先頭にタイトル見出しを付ける
  ;; すでに先頭が # タイトル なら重複して付けない
  (defun my/pandoc--prepend-markdown-title (file title)
    "Prepend '# TITLE' to markdown FILE if TITLE exists."
    (when (and title (file-exists-p file))
      (with-temp-buffer
        (insert-file-contents file)
        (goto-char (point-min))
        (unless (looking-at-p "^# ")
          (insert "# " title "\n\n")
          (write-region (point-min) (point-max) file nil 'silent)))))
#+end_src

*** export current org file
現在開いているOrgファイルを、指定した形式で書き出す共通関数と、そのラッパー関数群。
普段は下にある pdf/docx/markdown/txt 用の個別関数を呼べばよい。

#+begin_src emacs-lisp
  ;; 現在開いているOrgファイルを、指定形式で書き出す共通関数
  ;; pdf/docx/md/txt の各関数は、この共通関数を呼ぶだけにしてある
  ;; こうすることで、保存やpandoc呼び出しの処理を一箇所に集約できる
  (defun my/pandoc--export-org-file (format)
    "Export current Org file with pandoc.
  FORMAT must be one of: pdf, docx, md, txt."
    (unless (buffer-file-name)
      (error "This buffer is not visiting a file"))
    (unless (derived-mode-p 'org-mode)
      (error "This command is for org-mode buffers"))

    ;; ファイル内容を最新化してからpandocへ渡す
    (save-buffer)

    (let* ((src (buffer-file-name))
           (title (my/pandoc--org-title))
           (ext (pcase format
                  ('pdf  "pdf")
                  ('docx "docx")
                  ('md   "md")
                  ('txt  "txt")
                  (_ (error "Unsupported format: %s" format))))
           (out (my/pandoc--target-file src ext))
           (args
            (append
             (list src "-f" "org" "-o" out)
             (pcase format
               ;; PDFはエンジンやフォント、余白などの設定が必要
               ('pdf  (my/pandoc--pdf-args))
               ;; DOCXは特別な引数なしでよい
               ('docx nil)
               ;; MarkdownはGitHub Flavored Markdownで出力する
               ('md   (list "-t" "gfm"))
               ;; TXTはplain textとして出力する
               ('txt  (list "-t" "plain"))
               (_ nil)))))
      (my/pandoc--run args)
      ;; Markdown出力時は、Orgタイトルがあれば先頭に # タイトル を付与する
      (when (eq format 'md)
        (my/pandoc--prepend-markdown-title out title))
      (when my/pandoc-open-after-export
        (my/pandoc--open-file out))
      (message "%s created: %s" (upcase ext) out)
      out))

  ;; 現在のOrgファイルをPDF化する
  ;; 共有用や印刷確認用の出力を想定する
  (defun my/pandoc-org-to-pdf ()
    "Export current Org file to PDF with pandoc."
    (interactive)
    (my/pandoc--export-org-file 'pdf))

  ;; 現在のOrgファイルをDOCX化する
  ;; Wordでの共有や再編集を想定する
  (defun my/pandoc-org-to-docx ()
    "Export current Org file to DOCX with pandoc."
    (interactive)
    (my/pandoc--export-org-file 'docx))

  ;; 現在のOrgファイルをMarkdown化する
  ;; 共有用テキストやGit系のREADME用途を想定する
  ;; ox-mdは使わず、pandocに統一する
  (defun my/pandoc-org-to-markdown ()
    "Export current Org file to Markdown with pandoc."
    (interactive)
    (my/pandoc--export-org-file 'md))

  ;; 現在のOrgファイルをTXT化する
  ;; メール貼り付けや簡易共有向けのプレーンテキストを想定する
  (defun my/pandoc-org-to-txt ()
    "Export current Org file to plain text with pandoc."
    (interactive)
    (my/pandoc--export-org-file 'txt))
#+end_src

  # Setup emacsclient path
  export EMACSCLIENT_PATH="/ucrt64/local/bin/emacsclientw.exe"
  emacs-vterm() {
      local dir="${1:-$PWD}"

      if [[ ! -d "$dir" ]]; then
          echo "not a directory: $dir" >&2
          return 1
      fi

      # MSYS2のパスをWindowsのパスに変換
      dir="$(cygpath -m "$dir")"

      # Lisp 文字列用に最低限エスケープ
      local escape="$dir"
      escape="${escape//\\/\\\\}"
      escape="${escape//\"/\\\"}"

      "$EMACSCLIENT_PATH" -n --eval "(my/vterm-here \"$escape\")" >/dev/null
  }
